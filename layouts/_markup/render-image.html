{{- /* Image render hook. Rewrites root-relative /images/... destinations to honor the site baseURL (so they resolve correctly when the site is served from a subpath like /memoryhub-blog/). External and already-prefixed URLs are left alone. */ -}} {{- $dest := .Destination -}} {{- if and (strings.HasPrefix $dest "/") (not (strings.HasPrefix $dest "//")) -}} {{- /* Strip leading slash so relURL joins with the baseURL subpath (relURL leaves rooted paths unchanged otherwise). */ -}} {{- $dest = strings.TrimPrefix "/" $dest | relURL -}} {{- end -}}
